\section*{Objetivos da aula}
%==============================================================================
Ao final desta aula o aluno deve ser capaz de:
\begin{itemize}[leftmargin=1.4cm]
  \item explicar por que a \emph{acurácia} é insuficiente quando uma classe é rara ou
        quando os erros têm custos diferentes;
  \item ler uma \textbf{matriz de confusão} e definir sensibilidade, especificidade,
        TPR e FPR como estimativas de probabilidades;
  \item formular \textbf{perdas assimétricas} e deduzir o \emph{corte ótimo} de um
        classificador \emph{plug-in};
  \item construir e interpretar a \textbf{curva ROC} e a \textbf{AUC}.
\end{itemize}

\begin{emsala}
Continuação direta da Aula~07: lá vimos que o classificador de Bayes minimiza a
probabilidade de erro, mas a probabilidade de erro nem sempre é o que importa.
Pergunta de abertura: \emph{``numa população em que $1\%$ tem certa doença, um teste
que declara todo mundo saudável acerta $99\%$ das vezes. Ele é um bom teste?''}
\end{emsala}

%==============================================================================
\section{Por que a acurácia não basta}
%==============================================================================
Com $Y\in\{0,1\}$ e o risco $0$--$1$ da Aula~07, $\risco(g)=\Prob\big(g(\X)\neq Y\big)$,
a \textbf{acurácia} é $1-\risco(g)$. Ela trata todo erro como igual, o que falha em
dois casos:
\begin{itemize}[leftmargin=1.4cm,nosep]
  \item \textbf{custos assimétricos:} num rastreamento, um falso negativo costuma ser
        muito pior que um falso positivo;
  \item \textbf{classe rara:} com $\pi_1=\Prob(Y=1)$ pequena, $g\equiv0$ tem risco
        $\pi_1$ e acurácia $1-\pi_1$ sem acertar positivo nenhum. Se, além disso,
        $\Prob(Y=1\mid\X=\x)<1/2$ para todo $\x$, o próprio classificador de Bayes é
        $g\equiv0$ ([AME] §9.1).
\end{itemize}

\begin{atencao}
O desbalanceamento, sozinho, não é o problema: com custos iguais, o classificador de
Bayes continua minimizando o que interessa. O problema são os custos diferentes, e a
Seção~\ref{sec:custos} mostra como levá-los em conta.
\end{atencao}

\section{Matriz de confusão e métricas}
%==============================================================================
Fixado $g$, avaliamos numa amostra de teste $(\X_1,Y_1),\dots,(\X_m,Y_m)$, i.i.d.\ e
independente dos dados de treino, com $Y=1$ a classe positiva:
\begin{center}\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}ll cc@{}}
\toprule
 & & \multicolumn{2}{c}{\textbf{Verdadeiro}} \\
 & & $Y=0$ & $Y=1$ \\
\midrule
\multirow{2}{*}{\textbf{Predito}} & $g=0$ & VN (verd.\ neg.) & FN (falso neg.) \\
                                  & $g=1$ & FP (falso pos.) & VP (verd.\ pos.) \\
\midrule
 & total & $N$ & $P$ \\
\bottomrule
\end{tabular}
\end{center}
O \texttt{confusion\_matrix} do \texttt{scikit-learn} devolve a transposta desta
tabela, com o verdadeiro nas linhas. Os totais das colunas são os $P$ positivos e os
$N$ negativos verdadeiros. Cada medida estima uma probabilidade sobre uma observação
nova $(\X,Y)$:
\begin{align*}
  \text{sensibilidade:}\quad & S=\Prob\big(g(\X)=1\mid Y=1\big),
    & \widehat S&=\text{VP}/P,\\
  \text{especificidade:}\quad & E=\Prob\big(g(\X)=0\mid Y=0\big),
    & \widehat E&=\text{VN}/N.
\end{align*}
As duas condicionam na classe verdadeira e não dependem da prevalência
$\pi_1=\Prob(Y=1)$: com $g$ fixado, cada uma só depende da distribuição de $\X$
dentro de uma classe.

\begin{atencao}
Todas essas estimativas se fazem no conjunto de \emph{teste}, nunca no treino
(Aula~03).
\end{atencao}

Na ROC e no \texttt{scikit-learn} (a \texttt{roc\_curve} devolve \texttt{fpr} e
\texttt{tpr}), $S$ é a \textbf{taxa de verdadeiros positivos}, TPR, e $1-E$ é a
\textbf{taxa de falsos positivos}, FPR, estimadas por $\text{VP}/P$ e $\text{FP}/N$;
o complemento $1-S$ é a taxa de falsos negativos. Dados $P$ e $N$, a TPR e a FPR
reconstroem a matriz. Com $Y=0$ como hipótese nula e $g=1$ como rejeição, a FPR é a
probabilidade do erro do tipo~I e a TPR, o poder.

\begin{emsala}
O exemplo dos slides: 1000 pacientes, 10 doentes; o trivial $g_1\equiv0$ contra um
$g_2$ com VP $8$, FN $2$, FP $40$ e VN $950$. Pergunte qual dos dois o risco prefere
($0{,}010$ contra $0{,}042$: prefere $g_1$) e por quê: os $10$ doentes que $g_1$
perde são só $1\%$ dos pacientes, e os $40$ alarmes falsos de $g_2$, $4\%$. As taxas
por classe separam os dois erros: TPR $0$ contra $0{,}8$, FPR $0$ contra
$40/990\approx0{,}040$.
\end{emsala}

\section{Perdas assimétricas e o corte ótimo}\label{sec:custos}
%==============================================================================
Atribua custo $\ell_0>0$ a cada falso negativo (decidir $0$ quando $Y=1$) e
$\ell_1>0$ a cada falso positivo (decidir $1$ quando $Y=0$):
\begin{equation}\label{eq:riscoassim}
  \risco_\ell(g)=\ell_1\,\Prob\big(g(\X)=1,\,Y=0\big)
                +\ell_0\,\Prob\big(g(\X)=0,\,Y=1\big).
\end{equation}

\begin{proposicao}[Corte ótimo sob perda assimétrica]\label{prop:corte}
Com $\eta(\x)=\Prob(Y=1\mid\X=\x)$, o classificador
$g^*(\x)=\1{\ \eta(\x)>\tfrac{\ell_1}{\ell_0+\ell_1}\ }$ minimiza \eqref{eq:riscoassim}.
\end{proposicao}
\begin{proof}
Condicionando em $\X$,
\[
  \risco_\ell(g)=\E\Big[\,\ell_1\big(1-\eta(\X)\big)\1{g(\X)=1}
                +\ell_0\,\eta(\X)\1{g(\X)=0}\Big].
\]
Para cada $\x$, decidir $1$ custa $\ell_1\big(1-\eta(\x)\big)$ e decidir $0$ custa
$\ell_0\,\eta(\x)$; o primeiro é menor se e só se $\eta(\x)>\ell_1/(\ell_0+\ell_1)$.
Minimizar o integrando em cada $\x$ minimiza a esperança; no empate, qualquer decisão
é ótima.
\end{proof}

\begin{ideia}
Mudar os custos muda o corte, não o modelo: as probabilidades estimadas são as
mesmas. Com $\ell_0=\ell_1$ o corte é $1/2$ (Bayes); com $\ell_0=10\,\ell_1$, é
$1/11\approx0{,}09$. Só a razão $\ell_0/\ell_1$ importa.
\end{ideia}

\begin{emsala}
Pergunte: se deixar passar um doente custa dez vezes mais que um alarme falso, a
partir de que probabilidade se deve chamar alguém de positivo? ($1/11$, cerca de
$9\%$.)
\end{emsala}

\section{Escolhendo o corte: curva ROC e AUC}\label{sec:roc}
%==============================================================================
Um classificador \emph{plug-in} usa uma estimativa $\widehat\eta$ e um corte $p_0$:
$g_{p_0}(\x)=\1{\widehat\eta(\x)\ge p_0}$. Cada $p_0$ dá um par
$(\text{FPR}(p_0),\text{TPR}(p_0))$, e a \textbf{curva ROC} (\emph{Receiver Operating
Characteristic}) é o conjunto $\{(\text{FPR}(p_0),\text{TPR}(p_0)) : p_0\in[0,1]\}$,
a sensibilidade contra $1-$especificidade à medida que o corte varia. As duas taxas
diminuem ou ficam iguais quando $p_0$ aumenta: o conjunto dos classificados como
positivos só encolhe, e $P$ e $N$ não dependem do corte. Propriedades:
\begin{itemize}[leftmargin=1.4cm,nosep]
  \item vai de $(0,0)$ ($p_0$ acima de todas as probabilidades: ninguém é positivo) a
        $(1,1)$ ($p_0=0$: todos são);
  \item o canto superior esquerdo $(0,1)$ é o classificador sem erro;
  \item a diagonal é o ``palpite aleatório'': sortear $g=1$ com probabilidade $q$,
        ignorando $\x$, dá $\text{TPR}=\text{FPR}=q$;
  \item a \textbf{AUC} (área sob a curva) vale $1$ para o classificador perfeito, cuja
        curva passa por $(0,1)$, e $1/2$ para o sorteio da diagonal.
\end{itemize}

\begin{ideia}
Fixados os custos, o corte é o da Prop.~\ref{prop:corte}, com $\widehat\eta$ no lugar
de $\eta$; a curva mostra o que cada um dos outros cortes daria, e a AUC resume o
desempenho do modelo ao longo de todos eles.
\end{ideia}

\begin{atencao}
A curva ROC também deve ser construída no \emph{teste}: no treino, as duas taxas saem
otimistas.
\end{atencao}

%==============================================================================
\section*{Resumo}
%==============================================================================
\begin{itemize}[leftmargin=1.4cm]
  \item A acurácia engana com custos assimétricos ou classe rara; olhe a
        \textbf{matriz de confusão}.
  \item Sensibilidade (TPR) e especificidade ($1-\text{FPR}$) condicionam na classe
        verdadeira e não dependem da prevalência.
  \item \textbf{Perda assimétrica} \eqref{eq:riscoassim} $\Rightarrow$ corte ótimo
        $\ell_1/(\ell_0+\ell_1)$ (Prop.~\ref{prop:corte}).
  \item \textbf{ROC}: TPR contra FPR à medida que o corte varia; a \textbf{AUC} é a
        área sob ela.
\end{itemize}

\paragraph{Leitura recomendada.} [AME] §7.4 (matriz de confusão e métricas) e §9.1
(perdas assimétricas, cortes e a curva ROC); [ISLP] §4.4.2 (matriz de confusão, ROC
e AUC no exemplo do \texttt{Default}; Tabelas~4.6 e~4.7).
